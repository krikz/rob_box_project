    && rm -rf /var/lib/apt/lists/*

# Python-зависимости depthai (с --break-system-packages для Python 3.14)
RUN pip3 install --break-system-packages \
    opencv-python \
    depthai \
    && rm -rf /root/.cache/pip

# Инициализация rosdep (с guard)

RUN git clone --branch main https://github.com/luxonis/depthai-core.git && \
    cd depthai-core && \
    git checkout ${DEPTHAI_CORE_COMMIT} && \
    git submodule update --init --recursive

RUN git clone --branch main https://github.com/luxonis/depthai-ros.git && \
    cd depthai-ros && \
    git checkout ${DEPTHAI_ROS_COMMIT}

# Установка rosdep-зависимостей
WORKDIR /depthai_ws
RUN . /opt/ros/${ROS_DISTRO}/setup.sh && \
    rosdep install --from-paths src --ignore-src -r -y

# Сборка depthai-ros пакетов
# ВНИМАНИЕ: имена пакетов могут отличаться (hyphen vs underscore). Проверить после клонирования:
#   cd /depthai_ws && colcon list --names-only | grep depthai

    colcon build \
    --symlink-install \
    --cmake-args -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE} \
    --packages-select depthai-ros depthai_bridge depthai_ros_driver

RUN rm -rf /depthai_ws/src/depthai-core/.git && \
    rm -rf /depthai_ws/src/depthai-ros/.git && \
    rm -rf /depthai_ws/build/* && \
    rm -rf /depthai_ws/log/*

# Кэширование workspace setup
RUN echo "source /depthai_ws/install/setup.bash" >> /root/.bashrc

# Zenoh environment

    ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST \
    RUST_LOG=zenoh=info \
    LD_LIBRARY_PATH=/opt/ros/${ROS_DISTRO}/opt/zenoh_cpp_vendor/lib:/opt/ros/${ROS_DISTRO}/lib/aarch64-linux-gnu:/opt/ros/${ROS_DISTRO}/lib

# Точка входа — source обоих workspace
ENTRYPOINT ["/bin/bash", "-c", "source /opt/ros/lyrical/setup.bash && source /depthai_ws/install/setup.bash && exec \"$@\"", "--"]

КЛЮЧЕВЫЕ МОМЕНТЫ ДЛЯ ИСПОЛНИТЕЛЯ:
1. **Пиннинг коммитов (КРИТИЧНО):** Перед созданием Dockerfile выполнить:
   ```bash
   git ls-remote https://github.com/luxonis/depthai-core.git refs/heads/main | awk '{print $1}'
   git ls-remote https://github.com/luxonis/depthai-ros.git refs/heads/main | awk '{print $1}'
   ```
   Записать 40-char SHA в `DEPTHAI_CORE_COMMIT` и `DEPTHAI_ROS_COMMIT`.
2. **depthai пакет:** `pip install depthai` — Python SDK от Luxonis. Ставим с `--break-system-packages`. НЕ через apt (в Lyrical нет `ros-lyrical-depthai`).
3. **`libopencv-dev`:** Может быть конфликт с `ros:lyrical-ros-base` (в котором уже есть OpenCV). Если возникает → заменить на `ros-${ROS_DISTRO}-cv-bridge` (который тянет OpenCV как зависимость) и убрать `libopencv-dev`.
4. **depthai-core submodules:** `git submodule update --init --recursive` — обязателен. Без этого vcpkg и другие зависимости не скачаются.
5. **Сборочное время:** Первая сборка depthai-core на ARM64 может занять 2-4 часа (C++ компиляция + vcpkg). Последующие — минуты (кэш слоёв Docker).
6. **fallback:** Если сборка depthai-core падает на Resolute (GCC 15.2.0, vcpkg несовместимость) → использовать `luxonis/depthai-ros:kilted-arm64-latest` как временную базу (задокументировать в комментарии Dockerfile). ## Trust Boundaries

Миграция дистрибутива — code/config changes only. Новых trust boundaries не создаётся. Существующие границы:
- Docker build context → base image (начинается с ros:lyrical-ros-base из Docker Hub)
- Git clones → source code (navigation2, depthai-core, depthai-ros — все из официальных GitHub репозиториев)
- Docker image → runtime (запуск на роботе через docker-compose)

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-02-01 | Tampering | `git clone navigation2` | mitigate | Пиннинг SHA-коммита в ARG NAV2_COMMIT. Docker build падает если SHA не совпадает (git checkout fail). |
| T-05-02-02 | Tampering | `git clone depthai-core` | mitigate | Пиннинг SHA-коммита в ARG DEPTHAI_CORE_COMMIT. Аналогично git checkout validation. |
| T-05-02-03 | Tampering | `git clone depthai-ros` | mitigate | Пиннинг SHA-коммита в ARG DEPTHAI_ROS_COMMIT. |

| T-05-02-05 | Elevation of Privilege | `ENTRYPOINT` во всех Dockerfile'ах | accept | Контейнеры запускаются от root (ROS 2 Docker convention). Изоляция через Docker runtime + host network. |
| T-05-02-06 | Denial of Service | `colcon build` на ARM64 (2-4 часа) | accept | Длительная сборка — ожидаемо для source-build на ARM64. CI использует self-hosted runner с достаточными ресурсами. |
