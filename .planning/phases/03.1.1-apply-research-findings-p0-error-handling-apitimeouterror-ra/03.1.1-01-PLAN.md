async def _run_agent_with_retry(
    self,
    input_list: list,
    max_retries: int = 2,
    base_delay: float = 2.0,
):
    """Run Agent with retry on APIConnectionError (WiFi flakiness)."""
    last_exc = None
    for attempt in range(1 + max_retries):
        try:
            return await Runner.run(
                self._agent, input_list, max_turns=self._agent_max_turns
            )
        except APIConnectionError as exc:
            last_exc = exc
            if attempt < max_retries:
                delay = base_delay * (2**attempt)
                self.get_logger().warning(
                    f"⚠️ APIConnectionError (attempt {attempt + 1}/{1 + max_retries}), "
                    f"retrying in {delay:.0f}s: {exc}"
                )
                await asyncio.sleep(delay)
            else:
                self.get_logger().error(
                    f"❌ APIConnectionError after {1 + max_retries} attempts: {exc}"
                )
                raise
    raise last_exc
```

        for attempt in range(1 + max_retries):
            try:
                return await Runner.run(self._agent, input_list, max_turns=self._agent_max_turns)
            except APITimeoutError as exc:
                # Log timeout with request_id + underlying cause, then re-raise immediately
                # (no retry — timeouts are not transient network issues)
                self.get_logger().error(
                    f'⏰ LLM timeout after {self._llm_timeout}s '
                    f'(request_id={exc.request_id}, cause={exc.__cause__})'
                )
                raise
            except RateLimitError as exc:
                # Log rate limit, extract retry-after header, sleep, then retry (or give up)
                retry_after = exc.response.headers.get('retry-after', '5')
                self.get_logger().warn(
                    f'🔄 Rate limited (429), retry-after={retry_after}s '
                    f'(request_id={exc.request_id})'
                )
                await asyncio.sleep(float(retry_after) if retry_after.replace('.','').isdigit() else 5)
                # Falls through to next loop iteration (attempt counts as consumed)
            except APIStatusError as exc:
                # Log full API error with status code and truncated response body
                self.get_logger().error(
                    f'🌩️ API error {exc.status_code}: {exc.response.text[:200]} '
                    f'(request_id={exc.request_id})'
                )
                # Retry on 5xx (server errors) only; re-raise on 4xx (client errors)
                if exc.status_code >= 500 and attempt < max_retries:
                    await asyncio.sleep(base_delay * (2 ** attempt))
                    continue
                raise
            except APIConnectionError as exc:
                # ... existing WiFi flakiness handling — KEEP EXACTLY AS IS ...
    ```
