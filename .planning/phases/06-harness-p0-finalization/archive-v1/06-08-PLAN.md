```python
async def test_voice_input_to_tts_chain():
    """Full voice pipeline: STT text → DialogHarness → LLM → TTS side-effect → memory save."""
    harness = DialogHarness(config=...)
    harness.llm = DummyLLMProvider(responses=["Здравствуйте!"])
    harness.memory = InMemoryStore()
    harness.effects = RecordingBus()
    await harness.init()
    
    result = await harness.step("привет роббокс")
    
    # Verify LLM called with correct messages
    assert "Здравствуйте" in result
    # Verify TTS effect dispatched
    tts_effects = [e for e in harness.effects.effects if e.kind == "tts"]
    assert len(tts_effects) >= 1
    # Verify memory has both turns
    turns = await harness.memory.load_recent("user:default", limit=10)
    assert len(turns) >= 2
    
    await harness.teardown()
```
