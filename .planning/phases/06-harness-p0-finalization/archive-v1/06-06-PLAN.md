7. **TelegramHarness.step():**
```python
async def step(self, input_data: Any) -> str:
    # input_data is a TelegramUpdate-like dict: {chat_id, user_id, text, command, args}
    update = input_data if isinstance(input_data, dict) else {}
    self.state.chat_id = update.get("chat_id", "")
    self.state.user_id = update.get("user_id", "")
    self.state.message_count += 1
    
    # Auth check
    if not self._auth.check(self.state.user_id):
        return "unauthorized"
    
    # Command dispatch
    command = update.get("command", "")
    if command:
        self.state.last_command = command
        return await self._registry.dispatch(command, update.get("args", ""), self.state)
    
    # Text message → LLM
    text = update.get("text", "")
    if text:
        history = await self.memory.load_recent(f"tg:{self.state.chat_id}", limit=20)
        messages = [{"role": t.role, "content": t.content} for t in history]
        messages.append({"role": "user", "content": text})
        response = await self.llm.complete(messages, tools=[])
        await self.memory.append_turn(f"tg:{self.state.chat_id}", Turn(role="user", content=text))
        await self.memory.append_turn(f"tg:{self.state.chat_id}", Turn(role="assistant", content=response.content))
        return response.content
    
    return "no_action"
```
