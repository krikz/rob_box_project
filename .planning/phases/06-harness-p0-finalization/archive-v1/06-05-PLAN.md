4. **step() — main turn logic:**
```python
async def step(self, input_data: Any) -> str:
    # input_data is STT text string or InputEvent
    text = input_data if isinstance(input_data, str) else str(input_data)
    # Delegate to DialogueStateMachine for state transition
    event = self._dsm.on_user_input(text, self.state)
    if event == DialogueEvent.SILENCE_COMMAND:
        self.state.silenced = True
        return ""
    # Build messages from MemoryStore
    history = await self.memory.load_recent(f"user:{self.state.user_id}", limit=20)
    messages = [{"role": t.role, "content": t.content} for t in history]
    messages.append({"role": "user", "content": text})
    # Call LLM
    response = await self.llm.complete(messages, tools=self._skills.tool_specs())
    # Dispatch side-effects (TTS, LED, sound)
    await self.effects.dispatch(SideEffect(kind="tts", payload=response.content))
    # Save turn
    await self.memory.append_turn(f"user:{self.state.user_id}", Turn(role="user", content=text))
    await self.memory.append_turn(f"user:{self.state.user_id}", Turn(role="assistant", content=response.content))
    self.state.last_response = response.content
    self.state.turn_count += 1
    return response.content
```

```python
from enum import Enum, auto

class DialogueStateKind(Enum):
    IDLE = auto()        # waiting for wake word
    LISTENING = auto()   # wake word detected, listening for speech
    DIALOGUE = auto()    # processing user speech, LLM active
    SILENCED = auto()    # user said "тихо"/"молчи", temporarily suppressed
```

2. **Event enum:**
```python
class DialogueEvent(Enum):
    WAKE_WORD = auto()        # "роббокс" detected
    STT_RESULT = auto()       # speech recognized
    DIALOGUE_END = auto()     # LLM response complete
    SILENCE_COMMAND = auto()  # "тихо"/"молчи" detected
    TIMEOUT = auto()          # no activity timeout (return to IDLE)
    UNSILENCE = auto()        # silenced period expired
```

3. **DialogueStateMachine class:**
```python
class DialogueStateMachine:
    """Pure state machine — no I/O, no ROS2, no LLM.
    
    States: IDLE → LISTENING → DIALOGUE → IDLE
            any state → SILENCED → IDLE (after timeout)
    """
    
    def __init__(self, initial: DialogueStateKind = DialogueStateKind.IDLE, silence_timeout: float = 300.0):
        self._state = initial
        self._silence_timeout = silence_timeout  # seconds
        self._silenced_at: float = 0.0
    
    @property
    def state(self) -> DialogueStateKind:
        return self._state
    
    def on_event(self, event: DialogueEvent, dialog_state: DialogState) -> DialogueStateKind:
        """Process an event, return the new state."""
        # Transition table:
        # IDLE + WAKE_WORD → LISTENING
        # LISTENING + STT_RESULT → DIALOGUE
        # LISTENING + TIMEOUT → IDLE
        # DIALOGUE + DIALOGUE_END → IDLE
        # DIALOGUE + WAKE_WORD → LISTENING (barge-in)
        # any + SILENCE_COMMAND → SILENCED
        # SILENCED + TIMEOUT → IDLE (after silence_timeout)
        # SILENCED + UNSILENCE → IDLE
        ...
    
    def on_user_input(self, text: str, dialog_state: DialogState) -> DialogueEvent:
        """Classify user input into an event. Pure function — no side effects."""
        text_lower = text.lower().strip()
        if any(w in text_lower for w in ("роббокс", "роб бокс", "роб", "robbox")):
            return DialogueEvent.WAKE_WORD
        if any(w in text_lower for w in ("тихо", "молчи", "замолчи", "хватит", "стоп")):
            return DialogueEvent.SILENCE_COMMAND
        return DialogueEvent.STT_RESULT
```
